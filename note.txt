sql
SELECT
  to_timestamp(time_index / 1000.0) AS "time",
  neck_angle,
  trunk_angle,
  trunk_bending_angle,
  left_arm_angle,
  right_arm_angle,
  left_elbow_angle,
  right_elbow_angle,
  left_shoulder_angle,
  right_shoulder_angle,
  speed
FROM etergodata
WHERE to_timestamp(time_index / 1000.0) >= $__timeFrom()
  AND to_timestamp(time_index / 1000.0) <= $__timeTo()
ORDER BY time_index ASC
